## 부록 A. Firestore Rules (단계 0.5 시작 시 PR로 제출 예정)

```javascript
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // recipesss 전용
    match /recipeDrafts/{uid}/items/{draftId} {
      allow read, write: if request.auth != null && request.auth.uid == uid;
    }
    match /recipesssIngredients/{uid}/items/{ingId} {
      allow read, write: if request.auth != null && request.auth.uid == uid;
    }
    match /recipesssPresets/{uid}/items/{presetId} {
      allow read, write: if request.auth != null && request.auth.uid == uid;
    }
    match /recipesssSnapshots/{uid}/items/{snapId} {
      allow read, write: if request.auth != null && request.auth.uid == uid;
    }

    // 공유 (운영·생산앱과)
    match /nutrientProfiles/{profileId} {
      allow read: if request.auth != null;
    }
    match /usdaCache/{fdcId} {
      allow read, write: if request.auth != null;
    }
    match /recipes/{recipeId} {
      allow read: if request.auth != null;
      // write는 생산관리앱 Rules와 함께 조율 — DL-025
      allow write: if request.auth != null;
    }
  }
}
```
